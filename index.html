const { createApp, ref, computed, watch, nextTick } = Vue; const app = createApp({ setup() { // --- État --- const tasks = ref([]); const newTaskTitle = ref(''); const currentFilter = ref('all'); // Filtres disponibles const filters = [ { label: 'Toutes', value: 'all' }, { label: 'À faire', value: 'active' }, { label: 'Terminées', value: 'done' }, ]; // --- Propriétés calculées --- const filteredTasks = computed(() => { if (currentFilter.value === 'active') { return tasks.value.filter(t => !t.done); } if (currentFilter.value === 'done') { return tasks.value.filter(t => t.done); } return tasks.value; }); const hasCompleted = computed(() => { return tasks.value.some(t => t.done); }); // --- Méthodes --- // Ajouter une tâche const addTask = () => { const title = newTaskTitle.value.trim(); if (!title) return; tasks.value.push({ id: Date.now(), title: title, done: false, editing: false, }); newTaskTitle.value = ''; }; // Basculer l'état d'une tâche const toggleTask = (id) => { const task = tasks.value.find(t => t.id === id); if (task) { task.done = !task.done; } }; // Supprimer une tâche const deleteTask = (id) => { tasks.value = tasks.value.filter(t => t.id !== id); }; // Démarrer l'édition (double-clic) const startEdit = (task) => { // Désactiver l'édition sur toutes les autres tâches tasks.value.forEach(t => { if (t.id !== task.id) t.editing = false; }); task.editing = true; // Focus sur l'input après le render nextTick(() => { const input = document.querySelector( `[data-edit-id="${task.id}"]` ); if (input) { input.focus(); input.select(); } }); }; // Sauvegarder l'édition const saveEdit = (task) => { task.editing = false; // Si le titre est vide, on supprime la tâche if (!task.title.trim()) { deleteTask(task.id); } }; // Supprimer les tâches terminées const clearCompleted = () => { tasks.value = tasks.value.filter(t => !t.done); }; // Compter les tâches par filtre const getFilterCount = (filter) => { if (filter === 'all') return tasks.value.length; if (filter === 'active') { return tasks.value.filter(t => !t.done).length; } if (filter === 'done') { return tasks.value.filter(t => t.done).length; } return 0; }; // --- Watch : sauvegarde automatique dans localStorage --- watch( tasks, (newTasks) => { localStorage.setItem('tasks', JSON.stringify(newTasks)); }, { deep: true } ); // --- Chargement initial depuis localStorage --- const savedTasks = localStorage.getItem('tasks'); if (savedTasks) { try { tasks.value = JSON.parse(savedTasks); } catch (e) { console.warn('Erreur lors du chargement des tâches'); } } // --- Retour de l'API --- return { tasks, newTaskTitle, currentFilter, filters, filteredTasks, hasCompleted, addTask, toggleTask, deleteTask, startEdit, saveEdit, clearCompleted, getFilterCount, }; }, }); app.mount('#app');
